## Вертикальное и горизонтальное масштабирование

```bash
(
set -eu
for scale_node in a b; do
  if docker container inspect "pgcourse-scale-$scale_node" >/dev/null 2>&1; then
    printf 'Контейнер pgcourse-scale-%s уже существует.\n' "$scale_node" >&2
    exit 1
  fi
done
for scale_node in a b; do
  if [ "$scale_node" = a ]; then scale_port=5436; else scale_port=5437; fi
  install -d -o 999 -g 999 -m 0700 "/opt/pg-course/data/scale-$scale_node"
  install -d -o 999 -g 999 -m 0750 "/opt/pg-course/logs/scale-$scale_node"
  docker run -d --name "pgcourse-scale-$scale_node" --pull=never --restart=no \
    --label pg-course.lesson=scale --network pgcourse_course --log-driver none \
    --memory 256m --cpus 0.5 --shm-size 64m --user 999:999 \
    --publish "${scale_port}:5432" \
    --mount "type=bind,src=/opt/pg-course/data/scale-$scale_node,dst=/var/lib/postgresql" \
    --mount "type=bind,src=/opt/pg-course/logs/scale-$scale_node,dst=/var/log/course" \
    --mount type=bind,src=/opt/pg-course/certs/ca.crt,dst=/certs/ca.crt,readonly \
    --mount type=bind,src=/opt/pg-course/certs/server.crt,dst=/certs/server.crt,readonly \
    --mount type=bind,src=/opt/pg-course/certs/server.key,dst=/certs/server.key,readonly \
    --env-file /opt/pg-course/.env --env PGDATA=/var/lib/postgresql/18/docker \
    --entrypoint /bin/sh pg-course-local:18 -ec '
      umask 077
      exec >>/var/log/course/service.log 2>&1
      if [ ! -s "$PGDATA/PG_VERSION" ]; then
        printf "%s\n" "$POSTGRES_PASSWORD" >/tmp/scale-initdb-password
        initdb -D "$PGDATA" -U postgres --pwfile=/tmp/scale-initdb-password \
          --auth-local=trust --auth-host=scram-sha-256 \
          --encoding=UTF8 --locale=C.UTF-8
        printf "hostssl all all all scram-sha-256\n" >>"$PGDATA/pg_hba.conf"
        rm /tmp/scale-initdb-password
      fi
      exec postgres -D "$PGDATA" -c listen_addresses=0.0.0.0 \
        -c unix_socket_directories=/tmp -c shared_buffers=16MB \
        -c work_mem=1MB -c maintenance_work_mem=16MB -c max_connections=16 \
        -c wal_level=logical -c max_replication_slots=4 -c max_wal_senders=4 \
        -c max_slot_wal_keep_size=64MB -c max_wal_size=64MB -c min_wal_size=32MB \
        -c ssl=on -c ssl_cert_file=/certs/server.crt -c ssl_key_file=/certs/server.key \
        -c log_line_prefix="%m [%p] %u@%d app=%a "
    '
  for attempt in $(seq 1 30); do
    if docker exec --user postgres "pgcourse-scale-$scale_node" \
      pg_isready -q -h /tmp -U postgres -d postgres; then break; fi
    if [ "$(docker inspect --format '{{.State.Running}}' "pgcourse-scale-$scale_node")" != true ]; then break; fi
    sleep 1
  done
  docker exec --user postgres "pgcourse-scale-$scale_node" \
    pg_isready -h /tmp -U postgres -d postgres
  if [ "$(docker exec --user postgres "pgcourse-scale-$scale_node" \
    psql -X -At -h /tmp -U postgres -d postgres \
    -c "SELECT 1 FROM pg_database WHERE datname = 'scale';")" != 1 ]; then
    docker exec --user postgres "pgcourse-scale-$scale_node" \
      createdb -h /tmp -U postgres scale
  fi
done
)
```

## Шардирование и выбор владельца данных

```sql
WITH requests(tenant_id) AS (VALUES (10), (11), (12), (13))
SELECT tenant_id,
       CASE WHEN tenant_id % 2 = 0 THEN 'shard-a' ELSE 'shard-b' END AS target
FROM requests ORDER BY tenant_id;
```

```sql
CREATE SCHEMA app;
CREATE TABLE app.orders (
    tenant_id integer NOT NULL,
    order_id integer NOT NULL,
    amount numeric(12,2) NOT NULL CHECK (amount >= 0),
    PRIMARY KEY (tenant_id, order_id),
    CONSTRAINT orders_shard_owner CHECK (tenant_id % 2 = 0)
);
INSERT INTO app.orders VALUES (10, 1, 100), (10, 2, 150), (12, 1, 200);
SELECT tenant_id, order_id, amount FROM app.orders ORDER BY tenant_id, order_id;
```

## Логическая репликация выбранных данных

[русской документации логической репликации](https://postgrespro.ru/docs/postgresql/18/logical-replication)

```sql
CREATE TABLE app.catalog (
    market text NOT NULL,
    product_code text NOT NULL,
    product_name text NOT NULL,
    price numeric(12,2) NOT NULL CHECK (price >= 0),
    PRIMARY KEY (market, product_code)
);
INSERT INTO app.catalog VALUES
    ('ru', 'P1', 'Кружка', 100),
    ('ru', 'P2', 'Тарелка', 200),
    ('en', 'P1', 'Mug', 150);
SELECT market, product_code, product_name, price
FROM app.catalog ORDER BY market, product_code;
```

[русской документации фильтров строк](https://postgrespro.ru/docs/postgresql/18/logical-replication-row-filter)

```sql
CREATE ROLE scale_repl WITH LOGIN REPLICATION PASSWORD 'ScaleRepl-18-Lab!';
GRANT USAGE ON SCHEMA app TO scale_repl;
GRANT SELECT ON app.catalog TO scale_repl;
CREATE PUBLICATION scale_catalog_pub
FOR TABLE app.catalog WHERE (market = 'ru');
```

[русской документации особенностей логической репликации](https://postgrespro.ru/docs/postgresql/18/logical-replication-restrictions)

```sql
CREATE TABLE app.catalog (
    market text NOT NULL,
    product_code text NOT NULL,
    product_name text NOT NULL,
    price numeric(12,2) NOT NULL CHECK (price >= 0),
    PRIMARY KEY (market, product_code)
);
```

[русской документации CREATE SUBSCRIPTION](https://postgrespro.ru/docs/postgresql/18/sql-createsubscription)

```sql
CREATE SUBSCRIPTION scale_catalog_sub
CONNECTION 'host=192.168.0.35 port=5436 dbname=scale user=scale_repl password=ScaleRepl-18-Lab! sslmode=verify-full sslrootcert=/certs/ca.crt connect_timeout=5'
PUBLICATION scale_catalog_pub
WITH (slot_name = 'scale_catalog_slot', copy_data = true);
```

[русской документации каталога pg_subscription_rel](https://postgrespro.ru/docs/postgresql/18/catalog-pg-subscription-rel)

```sql
SELECT srrelid::regclass AS relation, srsubstate
FROM pg_subscription_rel
WHERE srsubid = (SELECT oid FROM pg_subscription WHERE subname = 'scale_catalog_sub');
SELECT market, product_code, product_name, price
FROM app.catalog ORDER BY market, product_code;
```

[русской документации pg_replication_slots](https://postgrespro.ru/docs/postgresql/18/view-pg-replication-slots)

```sql
SELECT slot_name, slot_type, active, restart_lsn, confirmed_flush_lsn,
       pg_wal_lsn_diff(pg_current_wal_lsn(), restart_lsn) AS retained_wal_bytes,
       wal_status
FROM pg_replication_slots WHERE slot_name = 'scale_catalog_slot';
```
