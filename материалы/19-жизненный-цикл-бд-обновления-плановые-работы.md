[maintenance.sh](19-lifecycle-demo/maintenance.sh)

```powershell
$remoteDir = '/opt/pg-course/generated/lifecycle19'
ssh -4 -T master3-root "install -d -m 0700 '$remoteDir'"
if ($LASTEXITCODE -ne 0) { throw 'Не удалось подготовить каталог на master3' }
scp -4 -- './материалы-преподавателя/19-lifecycle-demo/maintenance.sh' "master3-root:$remoteDir/maintenance.sh"
if ($LASTEXITCODE -ne 0) { throw 'Не удалось скопировать сценарий' }
```

