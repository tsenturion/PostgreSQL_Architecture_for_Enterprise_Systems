```sql
SET ROLE lecture09_owner;
GRANT USAGE ON SCHEMA security09 TO lecture09_reader, lecture09_editor;
GRANT SELECT (request_id, tenant_login, subject, status)
    ON security09.requests TO lecture09_reader;
GRANT UPDATE (status) ON security09.requests TO lecture09_editor;
ALTER TABLE security09.requests ENABLE ROW LEVEL SECURITY;
CREATE POLICY requests_read ON security09.requests FOR SELECT TO lecture09_reader
    USING (tenant_login = session_user);
CREATE POLICY requests_update ON security09.requests FOR UPDATE TO lecture09_editor
    USING (tenant_login = session_user) WITH CHECK (tenant_login = session_user);
RESET ROLE;
```

## Аудит изменений и регистрация попыток

[русской документации SECURITY DEFINER](https://postgrespro.ru/docs/postgresql/18/sql-createfunction)

```sql
SET ROLE lecture09_owner;
CREATE TABLE security09_audit.events (
    event_id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    event_time timestamptz NOT NULL DEFAULT clock_timestamp(),
    actor name NOT NULL,
    backend_pid integer NOT NULL,
    transaction_id text NOT NULL,
    application_name text,
    request_id integer NOT NULL,
    old_status text NOT NULL,
    new_status text NOT NULL
);
CREATE FUNCTION security09.audit_status() RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog AS $$
BEGIN
    INSERT INTO security09_audit.events
        (actor, backend_pid, transaction_id, application_name, request_id, old_status, new_status)
    VALUES (session_user, pg_backend_pid(), pg_current_xact_id()::text,
            current_setting('application_name', true), NEW.request_id, OLD.status, NEW.status);
    RETURN NEW;
END;
$$;
REVOKE ALL ON FUNCTION security09.audit_status() FROM PUBLIC;
CREATE TRIGGER audit_request_status AFTER UPDATE OF status ON security09.requests
FOR EACH ROW WHEN (OLD.status IS DISTINCT FROM NEW.status)
EXECUTE FUNCTION security09.audit_status();
GRANT USAGE ON SCHEMA security09_audit TO lecture09_auditor;
GRANT SELECT ON security09_audit.events TO lecture09_auditor;
RESET ROLE;
```
