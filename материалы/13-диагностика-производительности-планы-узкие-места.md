```sql
CREATE EXTENSION pg_stat_statements;
CREATE ROLE perf13_app LOGIN PASSWORD 'Perf13-client-2026!';
CREATE SCHEMA perf13;
CREATE TABLE perf13.tenants (
    tenant_id integer PRIMARY KEY,
    name text NOT NULL
);
INSERT INTO perf13.tenants
SELECT n, 'Организация ' || n FROM generate_series(1, 100) AS n;
CREATE TABLE perf13.orders (
    order_id bigint PRIMARY KEY,
    tenant_id integer NOT NULL REFERENCES perf13.tenants,
    state text NOT NULL,
    created_at timestamptz NOT NULL,
    amount numeric(12,2) NOT NULL,
    country text NOT NULL,
    currency text NOT NULL,
    payload text NOT NULL
);
INSERT INTO perf13.orders
SELECT n, 1 + (n - 1) % 100,
       CASE WHEN ((n - 1) / 100) % 5 = 0 THEN 'open' ELSE 'paid' END,
       timestamptz '2026-01-01 00:00:00+00' + n * interval '1 minute',
       (n % 10000)::numeric / 100,
       (ARRAY['RU', 'US', 'DE', 'FR'])[1 + (n - 1) % 4],
       (ARRAY['RUB', 'USD', 'EUR', 'EUR'])[1 + (n - 1) % 4],
       lpad(((n * 7919) % 1000003)::text, 12, '0') || repeat(chr(65 + n % 26), 116)
FROM generate_series(1, 60000) AS n;
CREATE TABLE perf13.skew (
    id integer PRIMARY KEY,
    lookup_key integer NOT NULL,
    payload text NOT NULL
);
INSERT INTO perf13.skew
SELECT n, CASE WHEN n <= 9900 THEN 1 ELSE n END, repeat('s', 64)
FROM generate_series(1, 10000) AS n;
CREATE INDEX skew_lookup_idx ON perf13.skew(lookup_key);
GRANT USAGE ON SCHEMA perf13 TO perf13_app;
GRANT SELECT ON ALL TABLES IN SCHEMA perf13 TO perf13_app;
```

## Повторное выполнение и изменение формы запроса

```sql
EXPLAIN (ANALYZE, BUFFERS, TIMING OFF)
SELECT t.tenant_id,
       COALESCE((SELECT sum(o.amount)
                 FROM perf13.orders o
                 WHERE o.tenant_id = t.tenant_id AND o.state = 'open'), 0) AS open_amount
FROM perf13.tenants t
WHERE t.tenant_id <= 10
ORDER BY t.tenant_id;
```

```sql
EXPLAIN (ANALYZE, BUFFERS, TIMING OFF)
SELECT t.tenant_id, COALESCE(sum(o.amount), 0) AS open_amount
FROM perf13.tenants t
LEFT JOIN perf13.orders o ON o.tenant_id = t.tenant_id AND o.state = 'open'
WHERE t.tenant_id <= 10
GROUP BY t.tenant_id
ORDER BY t.tenant_id;
```

```sql
WITH correlated AS (
    SELECT t.tenant_id,
           COALESCE((SELECT sum(o.amount) FROM perf13.orders o
                     WHERE o.tenant_id = t.tenant_id AND o.state = 'open'), 0) AS open_amount
    FROM perf13.tenants t WHERE t.tenant_id <= 10
), grouped AS (
    SELECT t.tenant_id, COALESCE(sum(o.amount), 0) AS open_amount
    FROM perf13.tenants t
    LEFT JOIN perf13.orders o ON o.tenant_id = t.tenant_id AND o.state = 'open'
    WHERE t.tenant_id <= 10 GROUP BY t.tenant_id
)
SELECT count(*) AS differences
FROM ((TABLE correlated EXCEPT ALL TABLE grouped)
      UNION ALL
      (TABLE grouped EXCEPT ALL TABLE correlated)) AS delta;
```

```bash
#!/usr/bin/env bash
set -Eeuo pipefail
cd /opt/pg-course
phase=${1:?Укажите фазу baseline, index или cpu}
case "$phase" in baseline|index|cpu) ;; *) exit 2 ;; esac
test ! -e "state/lecture13/$phase"
install -d -o 999 -g 999 -m 0700 "state/lecture13/$phase"
capture() {
  docker exec -i pgcourse-perf13 /opt/patroni/bin/python3 - <<'PY'
from pathlib import Path
import json
import time
base = Path('/sys/fs/cgroup')
cpu = {key: int(value) for key, value in (line.split() for line in (base / 'cpu.stat').read_text().splitlines())}
io = {'rbytes': 0, 'wbytes': 0}
for line in (base / 'io.stat').read_text().splitlines():
    fields = dict(field.split('=') for field in line.split()[1:])
    for key in io:
        io[key] += int(fields.get(key, 0))
print(json.dumps(dict(monotonic_ns=time.monotonic_ns(), cpu=cpu, io=io,
    cpu_max=(base / 'cpu.max').read_text().strip(),
    memory_bytes=int((base / 'memory.current').read_text()))))
PY
}
capture > "state/lecture13/$phase/start.json"
printf '\n%s lecture13 phase=%s\n' "$(date -u +%FT%TZ)" "$phase" \
  | tee -a logs/performance13/service.log
docker run --rm --name pgcourse-perf13-client --pull=never \
  --label pg-course.lesson=performance13 --network pgcourse_course --log-driver none \
  --user 999:999 --memory 128m --memory-swap 128m --cpus 0.5 \
  -e PGHOST=192.168.0.35 -e PGPORT=5436 -e PGDATABASE=course -e PGUSER=perf13_app \
  -e PGPASSWORD='Perf13-client-2026!' -e PGSSLMODE=verify-full -e PGSSLROOTCERT=/certs/ca.crt \
  -e PGOPTIONS='-c work_mem=2MB -c statement_timeout=5s' \
  --mount type=bind,src=/opt/pg-course/generated/lecture13,dst=/work,readonly \
  --mount type=bind,src=/opt/pg-course/state/lecture13,dst=/results \
  --mount type=bind,src=/opt/pg-course/certs/ca.crt,dst=/certs/ca.crt,readonly \
  --entrypoint pgbench pg-course-local:18 \
  -n -c 4 -j 2 -T 10 -R 80 -L 200 -M simple -r -l --random-seed=13 \
  --log-prefix="/results/$phase/transactions" -f /work/read.sql@4 -f /work/report.sql@1 \
  2>&1 | tee -a logs/performance13/service.log
capture > "state/lecture13/$phase/end.json"
```

```bash
python3 - <<'PY'
from pathlib import Path
import json
import math

root = Path('/opt/pg-course/state/lecture13')
summary = []
for phase in ('baseline', 'index', 'cpu'):
    folder = root / phase
    start, end = [json.loads((folder / name).read_text()) for name in ('start.json', 'end.json')]
    elapsed = (end['monotonic_ns'] - start['monotonic_ns']) / 1e9
    usage = (end['cpu']['usage_usec'] - start['cpu']['usage_usec']) / 1e6
    operations = {}
    for script, name in ((0, 'read'), (1, 'report')):
        values, skipped, failed = [], 0, 0
        for path in folder.glob('transactions.*'):
            for line in path.read_text().splitlines():
                fields = line.split()
                assert len(fields) >= 7, fields
                if int(fields[3]) != script:
                    continue
                if fields[2] == 'skipped':
                    skipped += 1
                elif fields[2].isdigit():
                    values.append(int(fields[2]) / 1000)
                else:
                    failed += 1
        values.sort()
        operations[name] = dict(completed=len(values), skipped=skipped, failed=failed,
            above_200ms=sum(value > 200 for value in values),
            p95_ms=round(values[math.ceil(.95 * len(values)) - 1], 3) if values else None)
    summary.append(dict(phase=phase, cpu_max=end['cpu_max'],
        cpu_seconds=round(usage, 3), average_cpu=round(usage / elapsed, 3),
        throttled_periods=end['cpu']['nr_throttled'] - start['cpu']['nr_throttled'],
        read_bytes=end['io']['rbytes'] - start['io']['rbytes'],
        written_bytes=end['io']['wbytes'] - start['io']['wbytes'],
        memory_end_mib=round(end['memory_bytes'] / 1048576, 1), operations=operations))
text = json.dumps(summary, ensure_ascii=False, indent=2)
(root / 'summary.json').write_text(text + '\n')
print(text)
PY
```

## Завершение и проверка результата

```bash
python3 - <<'PY'
from pathlib import Path
import json
import shutil
import subprocess

root = Path('/opt/pg-course').resolve(strict=True)
state = root / 'state/lecture13'
summary = json.loads((state / 'summary.json').read_text())
with (root / 'logs/performance13/service.log').open('a') as log:
    log.write('\nlecture13 summary=' + json.dumps(summary, ensure_ascii=False) + '\n')
info = json.loads(subprocess.check_output(['docker', 'inspect', 'pgcourse-perf13']))[0]
assert info['Config']['Labels'].get('pg-course.lesson') == 'performance13'
assert subprocess.run(['docker', 'inspect', 'pgcourse-perf13-client'], capture_output=True).returncode != 0
subprocess.run(['docker', 'stop', '--signal', 'SIGINT', '--timeout', '60', 'pgcourse-perf13'], check=True)
stopped = json.loads(subprocess.check_output(['docker', 'inspect', 'pgcourse-perf13']))[0]['State']
assert stopped['ExitCode'] == 0 and not stopped['OOMKilled'], stopped
subprocess.run(['docker', 'rm', 'pgcourse-perf13'], check=True)
for relative in ('data/performance13', 'state/lecture13', 'generated/lecture13'):
    target = root / relative
    assert not target.is_symlink() and target.resolve(strict=True) == root / relative, target
    assert root in target.resolve().parents, target
    shutil.rmtree(target)
print('Диагностический экземпляр и рабочие файлы удалены; журнал и сводка сохранены.')
PY
```
