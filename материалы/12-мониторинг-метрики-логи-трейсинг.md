## Прикладная операция и её инструментирование

```python
from contextlib import closing
from datetime import datetime, timezone
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import parse_qs, urlsplit
import json
import logging
import os
import re
import secrets
import signal
import threading
import time
import psycopg2

logging.basicConfig(filename='/var/log/course/service.log', level=logging.INFO, format='%(message)s')
lock = threading.Lock()
totals = {'ok': 0, 'error': 0}
bounds = (0.1, 0.5, 1.0, 2.5)
buckets = {bound: 0 for bound in bounds}
duration_sum = 0.0

def event(body, **fields):
    logging.info(json.dumps(dict(timestamp=datetime.now(timezone.utc).isoformat(),
        service='course-orders', environment='lab', body=body, **fields), ensure_ascii=False))

def attr(key, value):
    return {'key': key, 'value': {'stringValue': value}}

class Handler(BaseHTTPRequestHandler):
    def log_message(self, *_):
        pass

    def send(self, status, body, content_type='application/json', traceparent=None):
        data = body.encode('utf-8')
        self.send_response(status)
        self.send_header('Content-Type', content_type)
        self.send_header('Content-Length', str(len(data)))
        if traceparent:
            self.send_header('traceparent', traceparent)
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        global duration_sum
        url = urlsplit(self.path)
        if url.path == '/metrics':
            with lock:
                count = sum(totals.values())
                lines = ['# TYPE pgcourse_requests_total counter']
                for outcome, value in totals.items():
                    lines.append(f'pgcourse_requests_total{{operation="orders.lookup",outcome="{outcome}"}} {value}')
                lines.append('# TYPE pgcourse_request_duration_seconds histogram')
                for bound, value in buckets.items():
                    lines.append(f'pgcourse_request_duration_seconds_bucket{{le="{bound}"}} {value}')
                lines.extend([f'pgcourse_request_duration_seconds_bucket{{le="+Inf"}} {count}',
                    f'pgcourse_request_duration_seconds_count {count}',
                    f'pgcourse_request_duration_seconds_sum {duration_sum}'])
            self.send(200, '\n'.join(lines) + '\n', 'text/plain; version=0.0.4; charset=utf-8')
            return
        case = parse_qs(url.query).get('case', ['ok'])[0]
        if url.path != '/lookup' or case not in ('ok', 'slow', 'error'):
            self.send(404, json.dumps({'error': 'Маршрут не найден'}))
            return
        incoming = re.fullmatch(r'00-([0-9a-f]{32})-([0-9a-f]{16})-01', self.headers.get('traceparent', ''))
        valid = incoming and int(incoming[1], 16) and int(incoming[2], 16)
        trace_id = incoming[1] if valid else secrets.token_hex(16)
        root_id, db_id = secrets.token_hex(8), secrets.token_hex(8)
        started, wall_start = time.monotonic(), time.time_ns()
        db_start = time.time_ns()
        outcome, sqlstate, matches = 'ok', '00000', None
        error_details = {}
        try:
            with closing(psycopg2.connect(host='pool', port=6432, dbname='course', user='course_app',
                    password=os.environ['APP_PASSWORD'], sslmode='verify-full',
                    sslrootcert='/certs/ca.crt', connect_timeout=5)) as conn:
                with conn:
                    with conn.cursor() as cursor:
                        cursor.execute("SELECT set_config('application_name', %s, true)", ('lecture12/' + trace_id,))
                        cursor.execute('SELECT pg_sleep(%s), count(*) / %s FROM sales.orders WHERE order_id = %s',
                                       (0.65 if case == 'slow' else 0.02, 0 if case == 'error' else 1, 1))
                        matches = cursor.fetchone()[1]
        except Exception as error:
            outcome = 'error'
            sqlstate = getattr(error, 'pgcode', None) or type(error).__name__
            error_details = dict(error_type=type(error).__name__, error_message=str(error).splitlines()[0])
        db_end, wall_end = time.time_ns(), time.time_ns()
        duration = time.monotonic() - started
        status = 200 if outcome == 'ok' else 500
        with lock:
            totals[outcome] += 1
            duration_sum += duration
            for bound in bounds:
                buckets[bound] += duration <= bound
        db_attributes = [attr('db.system.name', 'postgresql'), attr('db.namespace', 'course'),
                         attr('db.operation.name', 'SELECT'), attr('server.address', 'pool')]
        if outcome == 'error':
            db_attributes.append(attr('error.type', sqlstate))
        span_status = {'code': 1 if outcome == 'ok' else 2}
        root_span = dict(traceId=trace_id, spanId=root_id, name='GET /lookup', kind=2, flags=1,
            startTimeUnixNano=str(wall_start), endTimeUnixNano=str(wall_end), status=span_status)
        if valid:
            root_span['parentSpanId'] = incoming[2]
        db_span = dict(traceId=trace_id, spanId=db_id, parentSpanId=root_id, name='SELECT sales.orders', kind=3, flags=1,
            startTimeUnixNano=str(db_start), endTimeUnixNano=str(db_end),
            attributes=db_attributes, status=span_status)
        otlp = {'resourceSpans': [{'resource': {'attributes': [attr('service.name', 'course-orders'),
                    attr('deployment.environment.name', 'lab')]},
                    'scopeSpans': [{'scope': {'name': 'course.manual'}, 'spans': [root_span, db_span]}]}]}
        event('Обращение завершено', trace_id=trace_id, span_id=root_id, outcome=outcome,
              sqlstate=sqlstate, duration_seconds=duration, http_status=status, **error_details)
        self.send(status, json.dumps(dict(case=case, matches=matches, trace_id=trace_id, otlp=otlp),
                  ensure_ascii=False, indent=2), traceparent=f'00-{trace_id}-{root_id}-01')

def stop(*_):
    raise SystemExit(0)

signal.signal(signal.SIGTERM, stop)
signal.signal(signal.SIGINT, stop)
event('Сервис запущен', port=8000)
try:
    with ThreadingHTTPServer(('0.0.0.0', 8000), Handler) as server:
        server.serve_forever()
finally:
    event('Сервис остановлен')
```

```sql
SELECT queryid, calls, rows,
       round(total_exec_time::numeric, 2) AS total_exec_ms,
       round(mean_exec_time::numeric, 2) AS mean_exec_ms,
       query
FROM pg_stat_statements
WHERE dbid = (SELECT oid FROM pg_database WHERE datname = 'course')
  AND userid = 'course_app'::regrole
  AND query LIKE '%pg_sleep%'
  AND query LIKE '%count(*) /%'
  AND query LIKE '%sales.orders%'
ORDER BY total_exec_time DESC;
```

```yaml
groups:
  - name: lecture12
    interval: 30s
    rules:
      - alert: CoursePostgresCollectionFailed
        expr: up{job="postgres"} == 0
        for: 1m
        labels:
          severity: warning
          service: course-orders
          environment: lab
          cluster: course
          owner: dba
        annotations:
          summary: Потерян сбор метрик PostgreSQL
          action: Проверить цель опроса, ошибку последнего scrape и журнал экспортёра.
      - alert: CourseOrdersErrorRatioHigh
        expr: |
          sum by (service, environment, cluster, owner)
            (rate(pgcourse_requests_total{job="orders12",outcome="error"}[5m]))
          /
          clamp_min(sum by (service, environment, cluster, owner)
            (rate(pgcourse_requests_total{job="orders12"}[5m])), 0.001)
          > 0.05
        for: 2m
        labels:
          severity: warning
        annotations:
          summary: Повышена доля ошибок сервиса заказов
          action: Выбрать окружение и интервал, найти ошибки сервиса по trace_id и проверить зависимость PostgreSQL.
```
