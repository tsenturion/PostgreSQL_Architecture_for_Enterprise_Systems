## Совместимость как обязательное условие

```sql
BEGIN;
CREATE TEMP TABLE choice22_worker (job_id integer) ON COMMIT PRESERVE ROWS;
INSERT INTO choice22_worker VALUES (42);
COMMIT;

BEGIN;
SELECT count(*) AS rows_in_next_transaction FROM pg_temp.choice22_worker;
DROP TABLE pg_temp.choice22_worker;
COMMIT;
```

## Отбор по ограничениям и сравнение вариантов

```sql
WITH variants(name, worker_session_ok, server_budget) AS (
    VALUES
        ('транзакционный пул для всех', false, 24),
        ('сеансовый пул для всех', true, 84),
        ('А: веб через пул, задания напрямую', true, 24),
        ('Б: веб через пул, задания через отдельный сеансовый пул', true, 24)
)
SELECT name, worker_session_ok, server_budget,
       server_budget <= 32 AS budget_ok,
       worker_session_ok AND server_budget <= 32 AS eligible
FROM variants
ORDER BY name;
```

[Параметры пулов PgBouncer](https://www.pgbouncer.org/config.html)

```sql
WITH scores(option, simplicity, central_control) AS (
    VALUES ('А', 5, 2), ('Б', 3, 5)
), priorities(simplicity_weight) AS (
    VALUES (0.70::numeric), (0.60::numeric), (0.40::numeric)
)
SELECT simplicity_weight, option,
       simplicity_weight * simplicity
           + (1 - simplicity_weight) * central_control AS weighted_score
FROM priorities CROSS JOIN scores
ORDER BY simplicity_weight DESC, weighted_score DESC, option;
```

## Риски внедрения и проверка ограничений

```sql
WITH scenarios(name, pools, web_per_pool, worker_direct) AS (
    VALUES
        ('штатный режим', 1, 20, 4),
        ('выкладка с перекрытием', 2, 20, 4),
        ('выкладка с распределённой квотой', 2, 10, 4)
)
SELECT name,
       pools * web_per_pool + worker_direct AS server_connections,
       32 - (pools * web_per_pool + worker_direct) AS remaining_budget
FROM scenarios
ORDER BY pools, web_per_pool DESC;
```
