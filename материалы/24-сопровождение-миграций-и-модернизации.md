# Сопровождение миграций и модернизация на базе PostgreSQL

## Граница волны и правила преобразования

```sql
CREATE TEMP TABLE mig24_source (
    wave text NOT NULL,
    order_id integer NOT NULL,
    cents integer NOT NULL,
    status_code text NOT NULL CHECK (status_code IN ('P', 'D')),
    PRIMARY KEY (wave, order_id)
);

CREATE TEMP TABLE mig24_target (
    wave text NOT NULL,
    order_id integer NOT NULL,
    amount numeric(12, 2) NOT NULL,
    status text NOT NULL CHECK (status IN ('paid', 'pending')),
    PRIMARY KEY (wave, order_id)
);

CREATE TEMP TABLE mig24_route (
    wave text PRIMARY KEY,
    phase text NOT NULL CHECK (phase IN ('source', 'paused', 'target')),
    first_target_order integer,
    CHECK (phase = 'target' OR first_target_order IS NULL)
);

INSERT INTO mig24_route VALUES ('A', 'source', NULL), ('B', 'source', NULL);
INSERT INTO mig24_source VALUES
    ('A', 101, 10000, 'P'), ('A', 102, 25000, 'D'),
    ('B', 201, 7000, 'P');
```

```sql
CREATE TEMP VIEW mig24_expected AS
SELECT wave, order_id, (cents / 100.0)::numeric(12, 2) AS amount,
       CASE status_code WHEN 'P' THEN 'paid' WHEN 'D' THEN 'pending' END AS status
FROM mig24_source;

INSERT INTO mig24_target (wave, order_id, amount, status)
SELECT wave, order_id, amount, status
FROM mig24_expected;

UPDATE mig24_target
SET amount = amount + CASE order_id WHEN 101 THEN 10 ELSE -10 END
WHERE wave = 'A' AND order_id IN (101, 102);
```

## Сверка пилотной волны

```sql
SELECT 'source' AS dataset, count(*) AS order_count, sum(amount) AS total
FROM mig24_expected WHERE wave = 'A'
UNION ALL
SELECT 'target', count(*), sum(amount)
FROM mig24_target WHERE wave = 'A';
```

```sql
CREATE TEMP VIEW mig24_diff AS
SELECT COALESCE(s.wave, t.wave) AS wave,
       COALESCE(s.order_id, t.order_id) AS order_id,
       CASE
           WHEN s.order_id IS NULL THEN 'target_only'
           WHEN t.order_id IS NULL THEN 'source_only'
           ELSE 'different_value'
       END AS difference,
       s.amount AS expected_amount,
       t.amount AS target_amount,
       s.status AS expected_status,
       t.status AS target_status
FROM mig24_expected AS s
FULL JOIN mig24_target AS t USING (wave, order_id)
WHERE s.order_id IS NULL OR t.order_id IS NULL
   OR (s.amount, s.status) IS DISTINCT FROM (t.amount, t.status);

SELECT * FROM mig24_diff WHERE wave = 'A' ORDER BY order_id;
```

## Остановка записи и допуск к переключению

```sql
INSERT INTO mig24_source (wave, order_id, cents, status_code)
SELECT wave, 103, 7000, 'P'
FROM mig24_route WHERE wave = 'A' AND phase = 'source';

UPDATE mig24_route SET phase = 'paused'
WHERE wave = 'A' AND phase = 'source'
RETURNING wave, phase;
```

```sql
UPDATE mig24_route AS route
SET phase = 'target'
WHERE route.wave = 'A' AND route.phase = 'paused'
  AND NOT EXISTS (SELECT 1 FROM mig24_diff AS d WHERE d.wave = route.wave)
RETURNING wave, phase;
```

```sql
INSERT INTO mig24_target AS t (wave, order_id, amount, status)
SELECT s.wave, s.order_id, s.amount, s.status
FROM mig24_expected AS s
JOIN mig24_route AS r ON r.wave = s.wave
WHERE s.wave = 'A' AND r.phase = 'paused'
ON CONFLICT (wave, order_id) DO UPDATE
SET amount = EXCLUDED.amount, status = EXCLUDED.status
WHERE (t.amount, t.status) IS DISTINCT FROM (EXCLUDED.amount, EXCLUDED.status)
RETURNING wave, order_id;

UPDATE mig24_route AS route
SET phase = 'target'
WHERE route.wave = 'A' AND route.phase = 'paused'
  AND NOT EXISTS (SELECT 1 FROM mig24_diff AS d WHERE d.wave = route.wave)
RETURNING wave, phase;
```

## Первая запись на цели и завершение волны

```sql
BEGIN;
UPDATE mig24_route SET phase = 'source'
WHERE wave = 'A' AND phase = 'target' AND first_target_order IS NULL
RETURNING wave, phase;
ROLLBACK;
```

```sql
WITH accepted AS (
    INSERT INTO mig24_target (wave, order_id, amount, status)
    SELECT wave, 104, 120.00, 'paid'
    FROM mig24_route WHERE wave = 'A' AND phase = 'target'
    RETURNING wave, order_id
)
UPDATE mig24_route AS route
SET first_target_order = COALESCE(route.first_target_order, accepted.order_id)
FROM accepted
WHERE route.wave = accepted.wave
RETURNING route.wave, route.phase, route.first_target_order;
```

```sql
UPDATE mig24_route SET phase = 'source'
WHERE wave = 'A' AND phase = 'target' AND first_target_order IS NULL
RETURNING wave, phase;

SELECT wave, phase, first_target_order FROM mig24_route ORDER BY wave;
SELECT wave, order_id, difference FROM mig24_diff WHERE wave = 'A';
```
